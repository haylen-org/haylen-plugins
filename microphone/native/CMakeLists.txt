# The desktop part of the Microphone plugin, a C++ library that the desktop player and the Windows and Linux apps load. The tooling builds the target named after the plugin, `microphone`, against the installed SDK of the engine, whose target `haylen::native` gives it `haylen/platform/native/HaylenNative.h` and nlohmann/json. Each system records its microphone through its own API: AVFoundation in Objective-C++ on macOS, WASAPI on Windows and ALSA on Linux.
cmake_minimum_required(VERSION 3.28...4.4)
project(microphone LANGUAGES CXX)

find_package(haylen CONFIG REQUIRED)

add_library(microphone Microphone.cpp MicrophonePlugin.cpp)
target_link_libraries(microphone PRIVATE haylen::native)
target_compile_features(microphone PRIVATE cxx_std_20)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(microphone PRIVATE MicrophoneDevice.mm)
  set_source_files_properties(MicrophoneDevice.mm PROPERTIES COMPILE_OPTIONS -fobjc-arc)
  target_link_libraries(microphone PRIVATE "-framework AVFoundation" "-framework Foundation")
elseif(WIN32)
  target_sources(microphone PRIVATE MicrophoneDeviceWindows.cpp)
  target_link_libraries(microphone PRIVATE ole32 runtimeobject)
else()
  find_package(Threads REQUIRED)
  find_package(ALSA REQUIRED)
  target_sources(microphone PRIVATE MicrophoneDeviceLinux.cpp)
  target_link_libraries(microphone PRIVATE Threads::Threads ALSA::ALSA)
endif()
set_target_properties(microphone PROPERTIES CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden OBJCXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
