# The desktop part of the Dialogs plugin, a C++ library that the desktop player and the Windows and Linux apps load. The tooling builds the target named after the plugin, `dialogs`, against the installed SDK of the engine, whose target `haylen::native` gives it `haylen/platform/native/HaylenNative.h` and nlohmann/json. Every system reads the requests and writes the answers with the same classes, and shows its own dialogs: the sheets of AppKit in Objective-C++ on macOS, the dialogs of the shell on Windows and GTK 3, which it loads when it runs, on Linux.
cmake_minimum_required(VERSION 3.28...4.4)
project(dialogs LANGUAGES CXX)

find_package(haylen CONFIG REQUIRED)

add_library(dialogs Dialogs.cpp DialogsAnswer.cpp DialogsPlugin.cpp DialogsRequest.cpp)
target_link_libraries(dialogs PRIVATE haylen::native)
target_compile_features(dialogs PRIVATE cxx_std_20)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(dialogs PRIVATE MacosDialogs.mm)
  set_source_files_properties(MacosDialogs.mm PROPERTIES COMPILE_OPTIONS -fobjc-arc)
  target_link_libraries(dialogs PRIVATE "-framework AppKit" "-framework UniformTypeIdentifiers")
elseif(WIN32)
  target_sources(dialogs PRIVATE WindowsDialogs.cpp)
  target_compile_definitions(dialogs PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(dialogs PRIVATE ole32 shell32 uuid)
else()
  find_package(Threads REQUIRED)
  target_sources(dialogs PRIVATE LinuxDialogs.cpp)
  target_link_libraries(dialogs PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
endif()
set_target_properties(dialogs PROPERTIES CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden OBJCXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
