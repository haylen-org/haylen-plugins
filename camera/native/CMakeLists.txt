# The desktop part of the Camera plugin, a C++ library that the desktop player and the Windows and Linux apps load. The tooling builds the target named after the plugin, `camera`, against the installed SDK of the engine, whose target `haylen::native` gives it `haylen/platform/native/HaylenNative.h` and nlohmann/json. Each system captures its cameras through its own API: AVFoundation in Objective-C++ on macOS, Media Foundation on Windows and Video4Linux on Linux.
cmake_minimum_required(VERSION 3.28...4.4)
project(camera LANGUAGES CXX)

find_package(haylen CONFIG REQUIRED)

add_library(camera Camera.cpp CameraPlugin.cpp)
target_link_libraries(camera PRIVATE haylen::native)
target_compile_features(camera PRIVATE cxx_std_20)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(camera PRIVATE MacosCamera.mm MacosCameraDevices.mm MacosCapture.mm)
  set_source_files_properties(MacosCamera.mm MacosCameraDevices.mm MacosCapture.mm PROPERTIES COMPILE_OPTIONS -fobjc-arc)
  target_link_libraries(camera PRIVATE "-framework AVFoundation" "-framework CoreMedia" "-framework CoreVideo" "-framework CoreGraphics" "-framework ImageIO" "-framework Foundation")
elseif(WIN32)
  target_sources(camera PRIVATE WindowsCamera.cpp WindowsCameraDevices.cpp WindowsCameraPermission.cpp WindowsCapture.cpp WindowsPhoto.cpp WindowsText.cpp WindowsWorker.cpp)
  target_compile_definitions(camera PRIVATE NOMINMAX)
  target_link_libraries(camera PRIVATE mfplat mf mfreadwrite mfuuid evr ole32 windowscodecs runtimeobject)
else()
  find_package(Threads REQUIRED)
  target_sources(camera PRIVATE LinuxCamera.cpp LinuxCameraDevices.cpp LinuxCapture.cpp LinuxDescriptor.cpp LinuxMapping.cpp)
  target_link_libraries(camera PRIVATE Threads::Threads)
endif()
set_target_properties(camera PROPERTIES CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden OBJCXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
